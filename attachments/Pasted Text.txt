<script>
    // Simple drag-and-drop movement for workspace nodes
    const nodes = document.querySelectorAll('.node');
    nodes.forEach(node => {
        let isDragging = false, startX, startY;
        node.addEventListener('mousedown', (e) => {
            isDragging = true;
            startX = e.clientX - node.offsetLeft;
            startY = e.clientY - node.offsetTop;
            node.style.zIndex = 1000;
        });
        document.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            node.style.left = (e.clientX - startX) + 'px';
            node.style.top = (e.clientY - startY) + 'px';
        });
        document.addEventListener('mouseup', () => { isDragging = false; node.style.zIndex = 1; });
    });
</script>
